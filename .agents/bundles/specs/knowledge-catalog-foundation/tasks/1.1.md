- [ ] **Step 2 (`src/app/db/sql/catalog.sql`)**: Create `src/app/db/sql/catalog.sql` with three named SQL statements:
  ```sql
  -- name: select-catalog-table-annotations
  SELECT
      LOWER(object_name) AS table_name,
      annotation_name,
      annotation_value
  FROM user_annotations_usage
  WHERE object_type = 'TABLE'
    AND column_name IS NULL
    AND object_name IN (
        'PRODUCT',
        'STORE',
        'STORE_PRODUCT_INVENTORY',
        'RESPONSE_CACHE',
        'EMBEDDING_CACHE',
        'SEARCH_METRIC'
    )
  ORDER BY object_name, annotation_name;

  -- name: select-catalog-column-metadata
  SELECT
      LOWER(c.table_name) AS table_name,
      LOWER(c.column_name) AS column_name,
      c.data_type AS data_type,
      c.nullable AS nullable,
      c.column_id AS column_id,
      a.annotation_name AS annotation_name,
      a.annotation_value AS annotation_value
  FROM user_tab_columns c
  LEFT JOIN user_annotations_usage a
      ON a.object_name = c.table_name
     AND a.object_type = 'TABLE'
     AND a.column_name = c.column_name
  WHERE c.table_name IN (
      'PRODUCT',
      'STORE',
      'STORE_PRODUCT_INVENTORY',
      'RESPONSE_CACHE',
      'EMBEDDING_CACHE',
      'SEARCH_METRIC'
  )
  ORDER BY c.table_name, c.column_id, a.annotation_name;

  -- name: select-catalog-index-annotations
  SELECT
      LOWER(object_name) AS index_name,
      annotation_name,
      annotation_value
  FROM user_annotations_usage
  WHERE object_type = 'INDEX'
  ORDER BY object_name, annotation_name;
  ```

  ```python
  @dataclass(frozen=True, slots=True)
  class CatalogSettings:
      """Google Cloud Knowledge Catalog and local OKF context configuration."""

      enabled: bool = True
      mode: Literal["auto", "live", "local", "off"] = "auto"
      project_id: str | None = None
      location: str = "us-central1"
      entry_group_id: str = "cymbal-coffee-oracle"
      glossary_id: str = "cymbal-coffee-glossary"
      snapshot_path: str = str(BASE_DIR / "db" / "fixtures" / "knowledge_catalog_context.json")
      context_budget: int = 4096
      timeout_seconds: float = 10.0

      @classmethod
      def from_env(cls) -> CatalogSettings:
          """Load Knowledge Catalog settings from environment variables."""
          raw_mode = _env_str("KNOWLEDGE_CATALOG_MODE", "auto").strip().lower()
          if raw_mode == "gcp":
              raw_mode = "live"
          mode: Literal["auto", "live", "local", "off"] = (
              raw_mode if raw_mode in {"auto", "live", "local", "off"} else "auto"
          )
          return cls(
              enabled=_env_bool("KNOWLEDGE_CATALOG_ENABLED", True),
              mode=mode,
              project_id=os.getenv("KNOWLEDGE_CATALOG_PROJECT_ID") or os.getenv("GOOGLE_CLOUD_PROJECT"),
              location=_env_str("KNOWLEDGE_CATALOG_LOCATION", "us-central1"),
              entry_group_id=_env_str("KNOWLEDGE_CATALOG_ENTRY_GROUP", "cymbal-coffee-oracle"),
              glossary_id=_env_str("KNOWLEDGE_CATALOG_GLOSSARY_ID", "cymbal-coffee-glossary"),
              snapshot_path=_env_str(
                  "KNOWLEDGE_CATALOG_SNAPSHOT_PATH",
                  str(BASE_DIR / "db" / "fixtures" / "knowledge_catalog_context.json"),
              ),
              context_budget=_env_int("KNOWLEDGE_CATALOG_CONTEXT_BUDGET", 4096),
              timeout_seconds=_env_float("KNOWLEDGE_CATALOG_TIMEOUT_SECONDS", 10.0),
          )
  ```
