  ```python
  # SPDX-FileCopyrightText: 2026 Google LLC
  # SPDX-License-Identifier: Apache-2.0

  """Google Cloud Knowledge Catalog and Oracle 26ai annotations grounding service."""

  import time
  from pathlib import Path
  from typing import Any

  import google.auth
  import httpx
  import structlog
  from google.auth.transport.requests import Request
  from sqlspec.adapters.oracledb import OracleAsyncDriver

  from app.cli._helpers.catalog import (
      DATAPLEX_V1_BASE_URL,
      build_default_catalog_snapshot,
      load_catalog_snapshot,
      render_lookup_context_yaml,
  )
  from app.config import db_manager
  from app.domain.system.schemas import (
      CatalogContextPayload,
      CatalogEntryList,
      CatalogEntrySummary,
      CatalogGlossaryTerm,
  )
  from app.lib.settings import CatalogSettings, get_settings

  __all__ = ("KnowledgeCatalogService",)

  logger = structlog.get_logger()

  _CORE_TABLE_IDS = {
      "oracle:cymbal-coffee:product",
      "oracle:cymbal-coffee:store",
      "oracle:cymbal-coffee:store_product_inventory",
  }


  class KnowledgeCatalogService:
      """Retrieves governed Knowledge Catalog context and Oracle 26ai annotations."""

      def __init__(
          self,
          settings: CatalogSettings | None = None,
          driver: OracleAsyncDriver | None = None,
      ) -> None:
          """Initialize the Knowledge Catalog service.

          Args:
              settings: Optional CatalogSettings override.
              driver: Optional active Oracle database driver for live annotation inspection.
          """
          self._settings = settings or get_settings().catalog
          self._driver = driver
          self._cached_snapshot: CatalogContextPayload | None = None

      def list_entries(self) -> CatalogEntryList:
          """Return all registered Knowledge Catalog entries and glossary terms.

          Returns:
              CatalogEntryList containing entries and glossary terms.
          """
          snapshot = self._get_snapshot()
          return CatalogEntryList(
              entries=snapshot.entries,
              glossary_terms=snapshot.glossary_terms,
              total=len(snapshot.entries),
              mode=snapshot.mode,
          )

      async def lookup_context(
          self,
          query: str = "",
          *,
          entry_ids: list[str] | None = None,
          enabled: bool | None = None,
      ) -> CatalogContextPayload:
          """Retrieve LLM-ready Knowledge Catalog context for a user query or entry list.

          Args:
              query: User natural language query or topic filter.
              entry_ids: Optional explicit list of entry IDs to retrieve.
              enabled: Optional per-turn override enabling or disabling catalog grounding.

          Returns:
              CatalogContextPayload with matched entries, glossary terms, and lookupContext YAML.
          """
          start_time = time.perf_counter()
          is_enabled = self._settings.enabled if enabled is None else enabled
          if not is_enabled or self._settings.mode == "off":
              return CatalogContextPayload(
                  enabled=False,
                  mode="disabled",
                  entries=[],
                  glossary_terms=[],
                  active_privacy_rules=[],
                  active_data_roles=[],
                  context_yaml="",
                  latency_ms=round((time.perf_counter() - start_time) * 1000, 2),
              )

          snapshot = self._get_snapshot()
          matched_entries = _match_entries(snapshot.entries, query=query, entry_ids=entry_ids)
          matched_terms = _match_glossary_terms(snapshot.glossary_terms, query=query)
          live_annotations = await self._fetch_oracle_annotations_summary()

          resolved_mode = "local"
          context_yaml = render_lookup_context_yaml(matched_entries, matched_terms)

          if self._should_try_live():
              try:
                  live_yaml = await self._fetch_live_lookup_context(matched_entries)
                  if live_yaml:
                      context_yaml = live_yaml
                      resolved_mode = "live"
              except Exception as exc:
                  await logger.awarning(
                      "Live Knowledge Catalog lookupContext failed; using local snapshot fallback",
                      error_type=type(exc).__name__,
                      detail=str(exc),
                  )

          privacy_rules = list(snapshot.active_privacy_rules)
          for rule in live_annotations.get("privacy_rules", []):
              if rule not in privacy_rules:
                  privacy_rules.append(rule)

          data_roles = list(snapshot.active_data_roles)
          for role in live_annotations.get("data_roles", []):
              if role not in data_roles:
                  data_roles.append(role)

          latency_ms = round((time.perf_counter() - start_time) * 1000, 2)
          return CatalogContextPayload(
              enabled=True,
              mode=resolved_mode,
              entries=matched_entries,
              glossary_terms=matched_terms,
              active_privacy_rules=privacy_rules,
              active_data_roles=data_roles,
              context_yaml=context_yaml,
              latency_ms=latency_ms,
          )

      def _get_snapshot(self) -> CatalogContextPayload:
          """Load and memoize the local Knowledge Catalog snapshot."""
          if self._cached_snapshot is not None:
              return self._cached_snapshot
          path = Path(self._settings.snapshot_path)
          if path.exists():
              self._cached_snapshot = load_catalog_snapshot(path)
          else:
              self._cached_snapshot = build_default_catalog_snapshot()
          return self._cached_snapshot

      def _should_try_live(self) -> bool:
          """Determine whether live GCP Knowledge Catalog API calls should be attempted."""
          if self._settings.mode == "live":
              return bool(self._settings.project_id)
          if self._settings.mode == "auto":
              return bool(
                  self._settings.project_id
                  and self._settings.project_id not in {"demo-project", "your-project-id"}
              )
          return False

      async def _fetch_live_lookup_context(self, entries: list[CatalogEntrySummary]) -> str:
          """Call POST https://dataplex.googleapis.com/v1/...:lookupContext."""
          credentials, project = google.auth.default(
              scopes=["https://www.googleapis.com/auth/cloud-platform"]
          )
          credentials.refresh(Request())
          project_id = self._settings.project_id or str(project or "")
          if not project_id:
              return ""

          location = self._settings.location
          entry_group = self._settings.entry_group_id
          resources = [
              (
                  f"projects/{project_id}/locations/{location}/entryGroups/{entry_group}/entries/"
                  f"{entry.entry_id.replace(':', '-')}"
              )
              for entry in entries[:10]
          ]
          url = f"{DATAPLEX_V1_BASE_URL}/projects/{project_id}/locations/{location}:lookupContext"
          headers = {
              "Authorization": f"Bearer {credentials.token}",
              "Content-Type": "application/json",
          }
          async with httpx.AsyncClient(timeout=self._settings.timeout_seconds) as client:
              response = await client.post(url, headers=headers, json={"resources": resources})
              response.raise_for_status()
              data = response.json()
              return str(data.get("context", ""))

      async def _fetch_oracle_annotations_summary(self) -> dict[str, list[str]]:
          """Query live Oracle 26ai USER_ANNOTATIONS_USAGE when a database driver is present."""
          if self._driver is None:
              return {"privacy_rules": [], "data_roles": []}
          try:
              rows = await self._driver.select(db_manager.get_sql("select-catalog-column-metadata"))
          except Exception:
              return {"privacy_rules": [], "data_roles": []}

          privacy_rules: list[str] = []
          data_roles: list[str] = []
          for row in rows:
              r: dict[str, Any] = dict(row) if not isinstance(row, dict) else row
              tbl = str(r.get("table_name") or r.get("TABLE_NAME") or "").lower()
              col = str(r.get("column_name") or r.get("COLUMN_NAME") or "").lower()
              name = str(r.get("annotation_name") or r.get("ANNOTATION_NAME") or "")
              val = str(r.get("annotation_value") or r.get("ANNOTATION_VALUE") or "")
              if name == "Privacy" and val:
                  privacy_rules.append(f"{tbl}.{col}: {val}")
              elif name == "Data_Role" and val:
                  data_roles.append(f"{tbl}.{col}: {val}")
          return {"privacy_rules": privacy_rules, "data_roles": data_roles}


  def _match_entries(
      entries: list[CatalogEntrySummary],
      *,
      query: str,
      entry_ids: list[str] | None,
  ) -> list[CatalogEntrySummary]:
      """Select matching catalog entries while always preserving core domain tables."""
      if entry_ids:
          requested = {item.strip().lower() for item in entry_ids if item.strip()}
          matched = [e for e in entries if e.entry_id.lower() in requested or e.object_name.lower() in requested]
          if matched:
              return matched

      tokens = {tok for tok in query.lower().replace("?", " ").replace(",", " ").split() if len(tok) >= 3}
      if not tokens:
          return [e for e in entries if e.entry_id in _CORE_TABLE_IDS] or list(entries[:3])

      scored: list[tuple[int, CatalogEntrySummary]] = []
      for entry in entries:
          haystack = " ".join([
              entry.entry_id,
              entry.display_name,
              entry.description,
              entry.business_overview,
              *entry.steward_guidelines,
              *(col.column_name for col in entry.columns),
          ]).lower()
          score = sum(2 if tok in entry.display_name.lower() else 1 for tok in tokens if tok in haystack)
          if entry.entry_id in _CORE_TABLE_IDS:
              score += 1
          if score > 0:
              scored.append((score, entry))

      scored.sort(key=lambda pair: pair[0], reverse=True)
      return [entry for _, entry in scored[:4]] or list(entries[:3])


  def _match_glossary_terms(
      terms: list[CatalogGlossaryTerm],
      *,
      query: str,
  ) -> list[CatalogGlossaryTerm]:
      """Return all glossary terms ordered by relevance to the query."""
      tokens = {tok for tok in query.lower().split() if len(tok) >= 3}
      if not tokens:
          return list(terms)
      scored = sorted(
          terms,
          key=lambda term: sum(
              1 for tok in tokens if tok in f"{term.display_name} {term.definition}".lower()
          ),
          reverse=True,
      )
      return list(scored)
  ```
