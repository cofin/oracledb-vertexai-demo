  ```python
  from __future__ import annotations

  from unittest.mock import AsyncMock

  import pytest
  from dishka import Provider, Scope, make_async_container
  from dishka.integrations.litestar import DishkaRouter, LitestarProvider, setup_dishka
  from litestar import Litestar
  from litestar.testing import AsyncTestClient

  from app.domain.system.controllers import CatalogController, controllers
  from app.domain.system.schemas import CatalogContext
  from app.domain.system.services import KnowledgeCatalogService

  pytestmark = pytest.mark.anyio

  CONTEXT = CatalogContext(
      mode="local",
      tables=["product", "store", "store_product_inventory"],
      content="- table: product\n",
      latency_ms=0.42,
  )


  async def test_catalog_context_endpoint_returns_service_context() -> None:
      catalog_service = AsyncMock(spec=KnowledgeCatalogService)
      catalog_service.lookup_context.return_value = CONTEXT
      provider = Provider(scope=Scope.APP)
      provider.provide(lambda: catalog_service, provides=KnowledgeCatalogService)
      container = make_async_container(LitestarProvider(), provider)
      app = Litestar(route_handlers=[DishkaRouter(path="", route_handlers=[CatalogController])])
      setup_dishka(container, app)

      async with AsyncTestClient(app=app) as client:
          response = await client.get("/api/catalog/context")
      await container.close()

      assert response.status_code == 200
      assert response.json() == {
          "mode": "local",
          "tables": ["product", "store", "store_product_inventory"],
          "content": "- table: product\n",
          "latency_ms": 0.42,
      }


  def test_catalog_controller_is_registered_as_mcp_tool() -> None:
      assert CatalogController in controllers
  ```

  ```python
  # SPDX-FileCopyrightText: 2026 Google LLC
  # SPDX-License-Identifier: Apache-2.0

  """Knowledge Catalog context endpoint shared by HTTP and MCP clients."""

  from litestar import get
  from litestar_security import PublicController

  from app.domain.system.schemas import CatalogContext
  from app.domain.system.services import KnowledgeCatalogService
  from app.lib.di import Inject

  __all__ = ("CatalogController",)


  class CatalogController(PublicController):
      """Serve governed Knowledge Catalog context for the Cymbal Coffee tables."""

      @get(
          path="/api/catalog/context",
          name="system.catalog.context",
          summary="Knowledge Catalog Context",
          description="Governed context for product, store, and store_product_inventory from Google Cloud Knowledge Catalog.",
          mcp_tool="lookup_knowledge_catalog",
      )
      async def get_catalog_context(self, catalog_service: Inject[KnowledgeCatalogService]) -> CatalogContext:
          """Return Knowledge Catalog context from the live catalog or the committed snapshot."""
          return await catalog_service.lookup_context()
  ```
