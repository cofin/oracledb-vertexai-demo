- [ ] **Step 3: Wire `[data-kc-toggle]` in `src/resources/main.js`**
  In `src/resources/main.js`, add `setKnowledgeCatalogGrounding` and handle clicks on `[data-kc-toggle]`:
  ```javascript
  const setKnowledgeCatalogGrounding = (enabled) => {
    const input = document.querySelector("[data-kc-input]")
    const button = document.querySelector("[data-kc-toggle]")
    const status = document.querySelector("[data-kc-status]")
    if (input instanceof HTMLInputElement) {
      input.value = enabled ? "true" : "false"
    }
    if (button instanceof HTMLButtonElement) {
      button.setAttribute("aria-pressed", String(enabled))
      button.textContent = enabled ? "ON" : "OFF"
      button.classList.toggle("border-sage/35", enabled)
      button.classList.toggle("bg-sage/15", enabled)
      button.classList.toggle("text-sage", enabled)
      button.classList.toggle("border-border", !enabled)
      button.classList.toggle("bg-surface", !enabled)
      button.classList.toggle("text-muted", !enabled)
    }
    if (status instanceof HTMLElement) {
      status.textContent = enabled
        ? "LookupContext + Oracle 26ai Annotations active"
        : "Ungrounded mode (catalog context omitted)"
    }
  }
  ```
  And in `document.body.addEventListener("click", (event) => { ... })`:
  ```javascript
  const kcToggle = event.target.closest("[data-kc-toggle]")
  if (kcToggle instanceof HTMLButtonElement) {
    const currentlyEnabled = kcToggle.getAttribute("aria-pressed") !== "false"
    setKnowledgeCatalogGrounding(!currentlyEnabled)
    return
  }
  ```
